export const UsingCoinSchema = Type.Object({
    coin: Type.String({ description: 'Coin symbol (btc, eth, bch, ...)', example: 'btc' }),

export const UsingPathSchema = Type.Object({
    path: Type.Union([Type.String(), Type.Array(Type.Number())], {
        description: 'derivation path of the network, its second level is the slip44 coin type',
        example: "m/84'/0'/0'",
    }),
    networkType: Type.Union([Type.Literal('bitcoin'), Type.Literal('ethereum')], {
        description: 'network family the path is looked up in',
        example: 'bitcoin',
    }),
});

## Get Coin Info

Returns information about a specified coin from the [coins.json](https://github.com/trezor/trezor-suite/blob/develop/packages/connect-data/files/coins.json) file. See [supported coins](/details/coins) for every accepted `coin` value.

```javascript
const result = await TrezorConnect.getCoinInfo(params);
```

### Params

<CommonParamsLink />

#### Using coin

<ParamsTable schema={UsingCoinSchema} />

#### Using path

<ParamsTable schema={UsingPathSchema} />

### Example

Get coin info for Bitcoin.

```javascript
TrezorConnect.getCoinInfo({
    coin: 'btc',
});
```

Get coin info of the bitcoin network a path belongs to.

```javascript
TrezorConnect.getCoinInfo({
    path: "m/84'/0'/0'",
    networkType: 'bitcoin',
});
```
